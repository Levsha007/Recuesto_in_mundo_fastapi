Шаг 3: Создание FastAPI приложения
Теперь создадим FastAPI приложение и добавим несколько эндпоинтов для работы с базой данных.

python
Copy
from fastapi import FastAPI, Depends, HTTPException
from sqlalchemy.orm import Session
from typing import List

app = FastAPI()

# Зависимости для получения сессии базы данных
def get_db():
    db = SessionLocal()
    try:
        yield db
    finally:
        db.close()

# Эндпоинт для создания пользователя
@app.post("/users/", response_model=User)
def create_user(user: User, db: Session = Depends(get_db)):
    db_user = User(**user.dict())
    db.add(db_user)
    db.commit()
    db.refresh(db_user)
    return db_user

# Эндпоинт для получения всех пользователей
@app.get("/users/", response_model=List[User])
def read_users(skip: int = 0, limit: int = 100, db: Session = Depends(get_db)):
    users = db.query(User).offset(skip).limit(limit).all()
    return users

# Эндпоинт для получения пользователя по ID
@app.get("/users/{user_id}", response_model=User)
def read_user(user_id: int, db: Session = Depends(get_db)):
    db_user = db.query(User).filter(User.id == user_id).first()
    if db_user is None:
        raise HTTPException(status_code=404, detail="User not found")
    return db_user

Для чего нужен FASTApi и каким образом его использовать и интегрировать в сайт, состоящий из HTML is и css

FASTApi — это современный веб-фреймворк для Python, который используется для создания API (Application Programming Interface) с высокой производительностью. Он позволяет разработчикам быстро создавать и развертывать веб-приложения и API с минимальными усилиями. FASTApi особенно полезен для создания RESTful API, которые могут взаимодействовать с клиентскими приложениями, включая веб-сайты, мобильные приложения и другие сервисы.

Для чего нужен FASTApi?
Высокая производительность: FASTApi построен на основе асинхронного сервера Starlette и ASGI, что позволяет ему обрабатывать большое количество запросов с высокой скоростью.
Автоматическая генерация документации: FASTApi автоматически генерирует интерактивную документацию для вашего API с использованием Swagger UI и ReDoc.
Простота использования: FASTApi имеет простой и интуитивно понятный синтаксис, что упрощает разработку и поддержку кода.
Поддержка асинхронного программирования: FASTApi поддерживает асинхронные функции, что позволяет эффективно использовать ресурсы сервера.

Пример использования Tortoise ORM
Установка Tortoise ORM:


pip install tortoise-orm
Создание модели и подключение к базе данных:


from fastapi import FastAPI
from tortoise.contrib.fastapi import register_tortoise
from tortoise.models import Model
from tortoise import fields

app = FastAPI()

class Item(Model):
    id = fields.IntField(pk=True)
    name = fields.CharField(max_length=255)
    description = fields.TextField()
    price = fields.FloatField()
    tax = fields.FloatField()

@app.post("/items/")
async def create_item(item: Item):
    await item.save()
    return item

@app.get("/items/{item_id}")
async def read_item(item_id: int):
    return await Item.get(id=item_id)

register_tortoise(
    app,
    db_url="sqlite://db.sqlite3",
    modules={"models": ["__main__"]},
    generate_schemas=True,
    add_exception_handlers=True,
)

from fastapi.responses import HTMLResponse
from fastapi.staticfiles import StaticFiles

from pydantic import BaseModel

app = FastAPI()

# Настройка шаблонизатора Jinja2
templates = Jinja2Templates(directory="templates")

# Настройка статических файлов
app.mount("/static", StaticFiles(directory="static"), name="static")

# Создание таблиц в базе данных
Base.metadata.create_all(bind=engine)

class ItemCreate(BaseModel):
    name: str
    description: str = None
    price: float
    tax: float = None

@app.get("/", response_class=HTMLResponse)
async def read_root(request: Request):
    return templates.TemplateResponse("index.html", {"request": request})

@app.get("/api/items/{item_id}")
async def read_item(item_id: int, db: Session = Depends(get_db)):
    return db.query(Item).filter(Item.id == item_id).first()

@app.post("/api/items/")
async def create_item(item: ItemCreate, db: Session = Depends(get_db)):
    db_item = Item(**item.dict())
    db.add(db_item)
    db.commit()
    db.refresh(db_item)
    return db_item

Извините за недоразумение. Давайте уточним: когда мы говорим о "статических файлах" в контексте веб-разработки, мы имеем в виду файлы, которые не изменяются на сервере и просто отправляются клиенту "как есть". Это включает в себя HTML, CSS, JavaScript, изображения и шрифты. Эти файлы могут содержать динамический контент и анимации, которые выполняются на стороне клиента (в браузере).

Как это работает
Статические файлы: Это файлы, которые сервер отправляет клиенту без изменений. Например, CSS-файлы, JavaScript-файлы, изображения и шрифты.
Динамический контент: Это контент, который генерируется или изменяется на сервере перед отправкой клиенту. Например, HTML-шаблоны, которые заполняются данными из базы данных.
Анимации: Анимации, созданные с помощью JavaScript (например, с использованием библиотеки anime.js), выполняются на стороне клиента. Эти анимации не делают файлы "статическими" в смысле, что они не изменяются на сервере.

а в чем разница саййта с fastapi от сайта без него?
Разница между сайтом с FastAPI и сайтом без него заключается в архитектуре, функциональности и производительности. Давайте рассмотрим эти различия подробнее.

1. Архитектура
Сайт без FastAPI:

Статический сайт: Обычно состоит из статических HTML, CSS и JavaScript файлов. Сервер просто отдает эти файлы клиенту.

Динамический сайт: Может использовать серверные технологии, такие как PHP, Node.js (Express), Django, Flask и т.д., для генерации HTML на стороне сервера.

Сайт с FastAPI:

Микросервисная архитектура: FastAPI позволяет создавать микросервисы, которые могут быть использованы для обработки запросов и взаимодействия с базой данных.

API-центричный подход: FastAPI создан для создания API, которые могут быть использованы для взаимодействия с фронтендом (например, через AJAX запросы).

2. Функциональность
Сайт без FastAPI:

Простая маршрутизация: Обычно используется для статических сайтов или сайтов с простой логикой на стороне сервера.

Ограниченная интерактивность: Может использовать JavaScript для добавления интерактивности, но взаимодействие с сервером обычно происходит через формы или AJAX запросы.

Сайт с FastAPI:

Расширенная маршрутизация: FastAPI позволяет создавать сложные маршруты и обработчики запросов.

Интерактивность и динамический контент: FastAPI может генерировать динамический контент на стороне сервера и отправлять его клиенту через API.

Взаимодействие с базой данных: FastAPI позволяет легко интегрировать базы данных и выполнять CRUD операции.

3. Производительность
Сайт без FastAPI:

Статический контент: Быстрый доступ к статическим файлам, но медленная генерация динамического контента на стороне сервера.

Ограниченная масштабируемость: Может быть сложно масштабировать динамические сайты, особенно если они используют тяжелые серверные технологии.

Сайт с FastAPI:

Высокая производительность: FastAPI построен на основе асинхронного сервера Starlette и ASGI, что позволяет ему обрабатывать большое количество запросов с высокой скоростью.

Масштабируемость: FastAPI легко масштабируется, особенно при использовании асинхронных операций и микросервисной архитектуры.

4. Разработка и поддержка
Сайт без FastAPI:

Простая разработка: Для статических сайтов разработка может быть очень простой.

Сложность поддержки: Динамические сайты могут быть сложны в поддержке, особенно если они используют устаревшие технологии.

Сайт с FastAPI:

Современный инструмент: FastAPI использует современные подходы к разработке, такие как асинхронное программирование и автоматическая генерация документации.

Простота поддержки: Благодаря простой и интуитивно понятной структуре, FastAPI легко поддерживать и расширять.

Примеры использования
Сайт без FastAPI:

Лендинги и статические сайты: Сайты, которые не требуют динамического контента и сложной логики на стороне сервера.

Простые блоги и визитки: Сайты, где контент обновляется редко и не требует сложной обработки данных.

Сайт с FastAPI:

Интерактивные веб-приложения: Приложения, которые требуют динамического контента и сложной логики на стороне сервера.

API для мобильных приложений: FastAPI идеально подходит для создания API, которые могут быть использованы мобильными приложениями.

Микросервисы: FastAPI позволяет легко создавать микросервисы, которые могут быть использованы в более крупных системах.

Заключение
FastAPI предоставляет мощные инструменты для создания высокопроизводительных и масштабируемых веб-приложений с динамическим контентом. Если ваш сайт требует сложной логики на стороне сервера, взаимодействия с базой данных и высокой производительности, FastAPI будет отличным выбором. Если же ваш сайт статичен и не требует сложной обработки данных, то использование FastAPI может быть избыточным.

from fastapi import FastAPI, Request, Depends
from fastapi.responses import HTMLResponse
from fastapi.staticfiles import StaticFiles
from fastapi.templating import Jinja2Templates
from pydantic import BaseModel
from database.database import engine, Base, SessionLocal
from models.models import Item

app = FastAPI()

# Настройка шаблонизатора Jinja2
templates = Jinja2Templates(directory="templates")

# Настройка статических файлов
app.mount("/static", StaticFiles(directory="static"), name="static")

# Создание таблиц в базе данных
Base.metadata.create_all(bind=engine)

class ItemCreate(BaseModel):
    name: str
    description: str = None
    price: float
    tax: float = None

def get_db():
    db = SessionLocal()
    try:
        yield db
    finally:
        db.close()

@app.get("/", response_class=HTMLResponse)
async def read_root(request: Request):
    return templates.TemplateResponse("index.html", {"request": request})

@app.get("/page1", response_class=HTMLResponse)
async def read_page1(request: Request):
    return templates.TemplateResponse("page1.html", {"request": request})

@app.get("/page2", response_class=HTMLResponse)
async def read_page2(request: Request):
    return templates.TemplateResponse("page2.html", {"request": request})

# Добавьте аналогичные эндпоинты для других страниц

@app.get("/api/items/{item_id}")
async def read_item(item_id: int, db: Session = Depends(get_db)):
    return db.query(Item).filter(Item.id == item_id).first()

@app.post("/api/items/")
async def create_item(item: ItemCreate, db: Session = Depends(get_db)):
    db_item = Item(**item.dict())
    db.add(db_item)
    db.commit()
    db.refresh(db_item)
    return db_item

а, тоесть если их не удалить то они заменятся на новыйе или нет?
Да, папки __pycache__ создаются автоматически при запуске Python-приложений. Если вы не удалите их, они останутся на месте, и новые файлы будут добавлены в эти папки при следующем запуске.

Как работает __pycache__
Создание папки:

Когда вы запускаете Python-приложение, интерпретатор Python создает папку __pycache__ в каждой директории, содержащей исходные файлы Python (.py).

Внутри этой папки создаются файлы с расширением .pyc, которые содержат скомпилированный байт-код.

Использование папки:

При следующем запуске приложения Python сначала проверяет, есть ли в папке __pycache__ файлы с байт-кодом для соответствующих модулей.

Если такие файлы есть и они актуальны (т.е. соответствуют текущей версии исходных файлов), Python загружает их, что ускоряет запуск приложения.

Если файлы не актуальны или отсутствуют, Python компилирует исходные файлы заново и сохраняет новые файлы в __pycache__.

Что происходит при удалении __pycache__
Удаление папки:

Если вы удалите папку __pycache__, она будет создана заново при следующем запуске приложения.

Python скомпилирует исходные файлы заново и сохранит новые файлы с байт-кодом в новой папке __pycache__.

Влияние на производительность:

Удаление папки __pycache__ может немного замедлить первый запуск приложения, так как Python будет компилировать все модули заново.

Однако это замедление обычно незначительно и не оказывает существенного влияния на общую производительность.

Заключение
Папки __pycache__ создаются автоматически и могут быть удалены без вреда для работы вашего приложения. Они будут созданы заново при следующем запуске. Удаление этих папок может быть полезно для очистки проекта от временных файлов.

async def read_form_dvg_abt(request: Request):
    return templates.TemplateResponse("form_dvg_abt.html", {"request": request})

async def read_form_hotel_brn(request: Request):
    return templates.TemplateResponse("form_hotel_brn.html", {"request": request})

async def read_login(request: Request):
    return templates.TemplateResponse("login.html", {"request": request})

from database.database import engine, Base, SessionLocal
from models.models import Item

app = FastAPI()

# Настройка шаблонизатора Jinja2
templates = Jinja2Templates(directory="templates")

# Настройка статических файлов
app.mount("/static", StaticFiles(directory="static"), name="static")

# Создание таблиц в базе данных
Base.metadata.create_all(bind=engine)

class ItemCreate(BaseModel):
    name: str
    description: str = None
    price: float
    tax: float = None

def get_db():
    db = SessionLocal()
    try:
        yield db
    finally:
        db.close()

@app.get("/", response_class=HTMLResponse)
async def read_root(request: Request):
    return templates.TemplateResponse("index.html", {"request": request})

@app.get("/aquariums", response_class=HTMLResponse)
async def read_aquariums(request: Request):
    return templates.TemplateResponse("aquariums.html", {"request": request})

@app.get("/dayving", response_class=HTMLResponse)
async def read_dayving(request: Request):
    return templates.TemplateResponse("dayving.html", {"request": request})

@app.get("/form_dvg", response_class=HTMLResponse)
async def read_form_dvg(request: Request):
    return templates.TemplateResponse("form_dvg.html", {"request": request})

@app.get("/form_dvg_abt", response_class=HTMLResponse)
async def read_form_dvg_abt(request: Request):
    return templates.TemplateResponse("form_dvg_abt.html", {"request": request})

@app.get("/form_hotel_brn", response_class=HTMLResponse)
async def read_form_hotel_brn(request: Request):
    return templates.TemplateResponse("form_hotel_brn.html", {"request": request})

@app.get("/login", response_class=HTMLResponse)
async def read_login(request: Request):
    return templates.TemplateResponse("login.html", {"request": request})

@app.get("/resultPage", response_class=HTMLResponse)
async def read_resultPage(request: Request):
    return templates.TemplateResponse("resultPage.html", {"request": request})

@app.get("/api/items/{item_id}")
async def read_item(item_id: int, db: Session = Depends(get_db)):
    return db.query(Item).filter(Item.id == item_id).first()

@app.post("/api/items/")
async def create_item(item: ItemCreate, db: Session = Depends(get_db)):
    db_item = Item(**item.dict())
    db.add(db_item)
    db.commit()
    db.refresh(db_item)
    return db_item

Проверка маршрутов в FastAPI
Убедитесь, что все маршруты в вашем файле main.py настроены правильно и соответствуют путям, которые вы вводите в браузере.

from fastapi import FastAPI, Request, Depends
from fastapi.responses import HTMLResponse
from fastapi.staticfiles import StaticFiles
from fastapi.templating import Jinja2Templates
from pydantic import BaseModel
from sqlalchemy.orm import Session
from database.database import engine, Base, SessionLocal
from models.models import Item

app = FastAPI()

# Настройка шаблонизатора Jinja2
templates = Jinja2Templates(directory="templates")

# Настройка статических файлов
app.mount("/static", StaticFiles(directory="static"), name="static")

# Создание таблиц в базе данных
Base.metadata.create_all(bind=engine)

class ItemCreate(BaseModel):
    name: str
    description: str = None
    price: float
    tax: float = None

def get_db():
    db = SessionLocal()
    try:
        yield db
    finally:
        db.close()

@app.get("/", response_class=HTMLResponse)
async def read_root(request: Request):
    return templates.TemplateResponse("index.html", {"request": request})

@app.get("/aquariums", response_class=HTMLResponse)
async def read_aquariums(request: Request):
    return templates.TemplateResponse("aquariums.html", {"request": request})

@app.get("/dayving", response_class=HTMLResponse)
async def read_dayving(request: Request):
    return templates.TemplateResponse("dayving.html", {"request": request})

@app.get("/form_dvg", response_class=HTMLResponse)
async def read_form_dvg(request: Request):
    return templates.TemplateResponse("form_dvg.html", {"request": request})

@app.get("/form_dvg_abt", response_class=HTMLResponse)
async def read_form_dvg_abt(request: Request):
    return templates.TemplateResponse("form_dvg_abt.html", {"request": request})

@app.get("/form_hotel_brn", response_class=HTMLResponse)
async def read_form_hotel_brn(request: Request):
    return templates.TemplateResponse("form_hotel_brn.html", {"request": request})

@app.get("/login", response_class=HTMLResponse)
async def read_login(request: Request):
    return templates.TemplateResponse("login.html", {"request": request})

@app.get("/resultPage", response_class=HTMLResponse)
async def read_resultPage(request: Request):
    return templates.TemplateResponse("resultPage.html", {"request": request})

@app.get("/api/items/{item_id}")
async def read_item(item_id: int, db: Session = Depends(get_db)):
    return db.query(Item).filter(Item.id == item_id).first()

@app.post("/api/items/")
async def create_item(item: ItemCreate, db: Session = Depends(get_db)):
    db_item = Item(**item.dict())
    db.add(db_item)
    db.commit()
    db.refresh(db_item)
    return db_item
